cmake_minimum_required(VERSION 3.15)
project(keypop CXX)

if(NOT MSVC)
  message(FATAL_ERROR "MSVC only")
endif()

# Built without the CRT so the exe runs on Windows 7 regardless of toolset version.
# /RTC, /GS and exceptions all need CRT support code.
string(REGEX REPLACE "/RTC[^ ]*" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
string(REGEX REPLACE "/EH[a-z-]*" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")

# Version from the nearest vX.Y.Z tag, 0.0.0 without one.
set(KEYPOP_VERSION "0.0.0-dev")
find_package(Git QUIET)
if(GIT_FOUND)
  execute_process(COMMAND ${GIT_EXECUTABLE} describe --tags --dirty --always --match "v[0-9]*"
                  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
                  OUTPUT_VARIABLE KEYPOP_DESCRIBE OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(KEYPOP_DESCRIBE MATCHES "^v")
    string(SUBSTRING "${KEYPOP_DESCRIBE}" 1 -1 KEYPOP_VERSION)
  elseif(KEYPOP_DESCRIBE)
    set(KEYPOP_VERSION "0.0.0-g${KEYPOP_DESCRIBE}")
  endif()
  # ponytail: reconfigures on commit/checkout only, re-run cmake after adding a tag to HEAD
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/.git/logs/HEAD)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS .git/logs/HEAD)
  endif()
endif()
string(REGEX MATCH "^([0-9]+)\\.([0-9]+)\\.([0-9]+)" _ "${KEYPOP_VERSION}")
set(KEYPOP_MAJOR ${CMAKE_MATCH_1})
set(KEYPOP_MINOR ${CMAKE_MATCH_2})
set(KEYPOP_PATCH ${CMAKE_MATCH_3})
configure_file(src/version.h.in version.h @ONLY)
message(STATUS "keypop version ${KEYPOP_VERSION}")

# src/res/app.ico is generated by scripts/make_icon.py
add_executable(keypop WIN32 src/main.cpp src/app.rc)
target_include_directories(keypop PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(keypop PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601)
target_compile_options(keypop PRIVATE /W4 /GS- /GR-)
target_link_options(keypop PRIVATE /NODEFAULTLIB /ENTRY:Entry /SUBSYSTEM:WINDOWS,6.01)
# PDB for Release too, shipped with the release so crashes can be symbolised.
target_compile_options(keypop PRIVATE $<$<CONFIG:Release>:/Zi>)
target_link_options(keypop PRIVATE $<$<CONFIG:Release>:/DEBUG /OPT:REF /OPT:ICF>)
target_link_libraries(keypop PRIVATE kernel32 user32 gdi32 shell32)
